\begin{frame}{The problem is not only the handler}
\begin{center}

\begin{columns}[T]
\begin{column}{0.68\textwidth}
\begin{itemize}\small
\item A fresh environment imports code before it can serve the first request.
\item Optional SDKs can dominate that path even when a request never uses them.
\item Deferring imports can move heavy work into the handler.
\end{itemize}

\begin{column}{0.30\textwidth}
\begin{alertblock}{Question}\small
Can we reduce startup loading without moving the same cost into user-visible latency?
\end{alertblock}
\end{column}
\end{columns}

\begin{frame}{How PCSO removes the shifted-latency cost}
\centering\includegraphics[width=0.70\textwidth]{diagrams/predictive_loop.png}

\begin{itemize}\small
\item Static analysis says what can be deferred.

\end{itemize}

\begin{frame}{PCSO prediction under the hood}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Online first-order Markov model}
For every observed transition $A \rightarrow B$, PCSO increments the count in row $A$, column $B$.
\[
P(B\mid A)=\frac{\mathrm{count}(A\rightarrow B)}{\sum_x\mathrm{count}(A\rightarrow x)}
\]
The highest-probability next endpoint is selected when its confidence is at least 0.35.
\end{block}
\begin{exampleblock}{Small example}
If $A\rightarrow B$ occurs 7 times and $A\rightarrow C$ occurs 3 times, PCSO predicts $B$ with probability 0.70.
\end{exampleblock}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{What happens next}
\begin{enumerate}\small
\item Observe the current endpoint and record the completed transition.
\item Predict the next endpoint from the updated transition row.
\item Import the predicted endpoint and its local dependency closure into the warm set.
\item On the next request, reuse the resident module instead of importing it on the critical path.
\end{enumerate}
\end{block}
\begin{alertblock}{Miss behavior}\small
If confidence is low or the request differs from the prediction, PCSO falls back to the V1 static-loading path; correctness is unchanged.
\end{alertblock}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Why PCSO improves on the true baseline}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{1. V1 / FaaSLight minimum improvement}
Static AST analysis removes optional model, OCR, and media modules from startup. V1 reduces blended code-loading time by 53.9\%.
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{alertblock}{2. PCSO: prediction + preloading}
PCSO adds endpoint-transition learning and warms the predicted dependency closure, avoiding the deferred-import penalty inside the handler.
\end{alertblock}
\end{column}
\end{columns}
\vspace{1em}
Together, these changes reduce the blended fleet from 35.29 ms to 15.48 ms per request; PCSO improves total latency by 56.1\%.
\vspace{0.6em}
\begin{exampleblock}{Why Tier 3 improves sharply from V1 to PCSO}
V1 defers the expensive OCR, ML, and media imports, but Tier 3 still pays for them inside the handler: 88.98 ms versus 89.47 ms for the baseline. PCSO predicts the next endpoint and preloads those heavy modules before the request, reducing Tier 3 latency to 43.52 ms (51.4\% below baseline).
\end{exampleblock}
\end{frame}
